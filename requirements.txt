# Direct runtime dependencies, pinned.
#
# Pinned rather than floated because this app subclasses framework internals:
# `app/middlewares/ThrottleRequestsMiddleware.py` extends Masonite's own
# middleware, and `expire_stale_attempts` in `app/rate_limiters.py` depends on
# how `RateLimiter` / the cache `FileDriver` store their counters. An unpinned
# `masonite` upgrade would silently change rate limiting rather than fail loudly.
#
# NOTE: the framework package is `masonite-framework`, not `masonite`, and it
# pulls its ORM as `masonite-framework-orm`. A stale `masonite-orm` 2.24.0 may
# also be present in older virtualenvs -- both distributions install into the
# same `masoniteorm/` directory, so do NOT `pip uninstall masonite-orm` in an
# existing venv; it deletes files 3.1.0 still owns. Rebuild the venv instead.

masonite-framework==5.4.1
masonite-framework-orm==3.1.0

# Database driver (config/database.py, DB_CONNECTION=mysql).
PyMySQL==1.1.2

# Archives: PyMuPDF imports as `fitz` (app/services/ArchiveServices.py).
PyMuPDF==1.27.2.2

# Image derivatives + uploads. 12.3.0 for the 17 decoder advisories against
# 11.0.0 (pip-audit, SECURITY.md W6) -- it parses every uploaded image.
pillow==12.3.0

# Magic-byte upload validation (app/services/FileVerificationService.py).
python-magic==0.4.27

# News body sanitisation on write (NewsController._sanitize_news_html).
# 6.4.0 for GHSA-gj48-438w-jh9v and GHSA-8rfp-98v4-mmr6: this is the XSS filter.
bleach==6.4.0

# Realtime broadcasts; resolved by name through config/broadcast.py, so it is
# never imported directly and pip-freeze diffing will not flag it as missing.
pusher==3.3.4

# ROS bridge for the wayfinding IMU/GPS fusion (app/services/FusionServices.py).
roslibpy==2.1.0

# Datetime handling used directly in controllers and services, not only by
# the framework.
pendulum==3.2.0

# WSGI server used by lyn.sh in production.
gunicorn==25.3.0
